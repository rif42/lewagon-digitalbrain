  function catColor(cat){
    var c=(cat||'').toLowerCase();
    if(c.includes('dining')||c.includes('culinary')||c.includes('brunch')||c.includes('talk')) return '#284b63';

  function closeModal(){
    var m=document.getElementById('bali-event-modal');
    if(!m) return;
    m.hidden=true;
    document.body.style.overflow='';
  }
  function openModal(ev){
    var p=ev.extendedProps||ev;
    var modal=document.getElementById('bali-event-modal');
    if(!modal) return;
    document.getElementById('bali-modal-title').textContent=ev.title||p.name||'';
    var when='';
    if(p.start) { try{ var d=new Date(p.start); when=d.toLocaleString('en-GB',{dateStyle:'medium',timeStyle:'short',timeZone:'Asia/Makassar'})+' WITA'; }catch(e){ when=p.start; } }
    if(p.finish) { try{ var d2=new Date(p.finish); when+=' — '+d2.toLocaleString('en-GB',{dateStyle:'medium',timeStyle:'short',timeZone:'Asia/Makassar'})+' WITA'; }catch(e){ when+=' — '+p.finish; } }
    document.getElementById('bali-modal-meta').textContent=(p.category||'')+(p.location?' · '+p.location:'')+(when?' · '+when:'');
    document.getElementById('bali-modal-desc').textContent=p.description||'';
    document.getElementById('bali-modal-cost').textContent=p.cost?('Cost: '+p.cost):'';
    var a=document.getElementById('bali-modal-source');
    if(p.source){ a.href=p.source; a.textContent=p.source; a.hidden=false; } else { a.hidden=true; }
    var anchor=document.getElementById('bali-modal-anchor');
    if(p.anchor){ anchor.href='#'+p.anchor; anchor.hidden=false; } else { anchor.hidden=true; }
    modal.hidden=false;
    document.body.style.overflow='hidden';
  }
  function bindModalClose(){
    var closeBtn=document.getElementById('bali-modal-close');
    if(closeBtn && !closeBtn.dataset.baliBound){ closeBtn.addEventListener('click', function(e){ e.preventDefault(); closeModal(); }); closeBtn.dataset.baliBound='1'; }
    var modal=document.getElementById('bali-event-modal');
    if(modal && !modal.dataset.baliBound){
      modal.addEventListener('click',function(e){ if(e.target===modal) closeModal(); });
      modal.dataset.baliBound='1';
    }
    if(!document.body.dataset.baliEscBound){
      document.addEventListener('keydown',function(e){ if(e.key==='Escape'){ var m=document.getElementById('bali-event-modal'); if(m && !m.hidden) closeModal(); }});
      document.body.dataset.baliEscBound='1';
    }
  }
  function renderWeekly(events){
    var strip=document.getElementById('bali-weekly-strip');
    if(!strip) return;
    var weekly=events.filter(function(e){return e.recurring;});
    if(!weekly.length){ strip.hidden=true; return; }
    var html='<h4>Weekly in window — no fixed date</h4><ul>';
    weekly.forEach(function(w){
      html+='<li><strong>'+w.name+'</strong> <span class="wk-cat">('+w.category+')</span> — '+w.schedule+' · '+w.location+' · '+w.cost+' <a href="'+w.source+'" target="_blank" rel="noopener">source</a></li>';
    });
    html+='</ul>';
    strip.innerHTML=html;
  }
